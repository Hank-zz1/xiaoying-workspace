def handler(params):

    brands = plan.get('brands', [])

    if not brands:
        brands = fallback_brands(region)

    if not brands:

    all_results = []
    for brand_info in brands:
        brand_name = brand_info.get('name', '未知品牌')
        tasks = brand_info.get('tasks', [])
        brand_result = {"brand": brand_name, "region": region, "scrape": None, "ecommerce": None, "news": None}

        all_results.append(brand_result)

    return {"searchResult": json.dumps(all_results, ensure_ascii=False)}


def _ctx():
    c = ssl.create_default_context()
    c.check_hostname = False
    c.verify_mode = ssl.CERT_NONE
    return c

def mcp_call(mcp_url, tool, args, timeout=60):
    body = json.dumps({
        "jsonrpc": "2.0",
        "id": 1,
        "method": "tools/call",
        "params": {"name": tool, "arguments": args}
    }).encode('utf-8')

    req = urllib.request.Request(mcp_url, data=body, method='POST')
    req.add_header('Content-Type', 'application/json')
    req.add_header('X-Platform-Access-Key', ACCESS_KEY)

    try:
        with urllib.request.urlopen(req, context=_ctx(), timeout=timeout) as resp:
            raw = resp.read().decode('utf-8')
    except urllib.error.HTTPError as e:
        return {"error": f"HTTP {e.code}: {str(e)[:200]}"}
    except Exception as e:
        return {"error": str(e)[:200]}

    if 'event:' in raw or 'data:' in raw:
        for line in raw.split('\n'):
            if line.startswith('data:'):
                data_str = line[5:].strip()
                if data_str:
                    try:
                        return json.loads(data_str)
                    except:
                        continue

    try:
        return json.loads(raw)
    except:
        return {"error": "无法解析响应", "raw": raw[:500]}


def is_blocked(md):

        'to discuss automated access', 'dogsofamazon', 'error/500_503',
        'error/logo', '500 internal server error',
    ]
    md_lower = md.lower()
    return any(sig in md_lower for sig in blocked_sigs)


def fc_scrape(url):
    resp = mcp_call(FC_MCP, "firecrawl_scrape", {"url": url, "formats": ["markdown"], "waitFor": 3000})
    if isinstance(resp, dict):
        data = resp.get('data', {})
        if isinstance(data, dict) and data.get('markdown'):
            return data['markdown']
        if resp.get('markdown'):
            return resp['markdown']
        content = resp.get('content', [])
        if content and isinstance(content, list):
            try:
                inner = json.loads(content[0].get('text', '{}'))
                if isinstance(inner, dict):
                    d = inner.get('data', inner)
                    return d.get('markdown', '') or ''
            except:
                pass
        result = resp.get('result', {})
        if isinstance(result, dict):
            rcontent = result.get('content', [])
            if rcontent and isinstance(rcontent, list):
                try:
                    inner = json.loads(rcontent[0].get('text', '{}'))
                    if isinstance(inner, dict):
                        d = inner.get('data', inner)
                        return d.get('markdown', '') or ''
                except:
                    pass
    return ""


def bocha_search(query, count=5):
    """使用博查搜索：先 AI 搜索，空则回退 web 搜索"""
    articles = []

    def _extract_bocha(resp):
        """博查 MCP 响应经过 JSON-RPC 包装: result.content[0].text 是实际 JSON 字符串"""
        if not isinstance(resp, dict):
            return []

        # 先解包 JSON-RPC: result.content[0].text
        result = resp.get('result', {})
        if isinstance(result, dict):
            content = result.get('content', [])
            if content and isinstance(content, list):
                for item in content:
                    if item.get('type') == 'text' and item.get('text'):
                        try:
                            resp = json.loads(item['text'])
                        except:
                            pass

        msgs = resp.get('messages', [])
        if not msgs:
            return []
        for msg in msgs:
            if msg.get('content_type') != 'webpage':
                continue
            raw = msg.get('content', '')
            if not raw:
                continue
            try:
                inner = json.loads(raw)
                if isinstance(inner, dict) and 'value' in inner:
                    return inner['value']
            except:
                pass
        return []

    def _collect(pages):
        for p in pages[:count]:
            if isinstance(p, dict):
                articles.append({
                    "title": (p.get('name', '') or p.get('title', '') or '').replace('\\', ''),
                    "snippet": (p.get('snippet', '') or p.get('summary', '') or p.get('description', '') or '')[:200],
                    "url": p.get('url', ''),
                    "source": p.get('siteName', '') or p.get('source', '')
                })

    # 先试 ai_search
    resp = mcp_call(BOCHA_MCP, "ai_search", {"query": query, "count": count, "answer": False})
    _collect(_extract_bocha(resp))

    # ai_search 空则回退 web_search
    if not articles:
        resp2 = mcp_call(BOCHA_MCP, "web_search", {"query": query, "count": count})
        _collect(_extract_bocha(resp2))

    return articles

    def fallback_search():

        if fallback_search_cn:
            articles = bocha_search(fallback_search_cn, 5)
            if articles:
                return articles
        articles = bocha_search(f"{brand_name} refrigerator 2026 new models", 5)
        return articles

    if not url:
        articles = fallback_search()
        if articles:
            models = extract_models_from_articles(articles)
            return {"status": "fallback", "models": models, "model_count": len(models), "articles": articles}
        result["error"] = "无可用官网URL"
        return result

    try:
        md = fc_scrape(url)
        if not md:
            articles = fallback_search()
            if articles:
                models = extract_models_from_articles(articles)
                return {"status": "fallback", "models": models, "model_count": len(models), "articles": articles}
            result["error"] = "官网抓取失败：无内容返回"
            return result

        if is_blocked(md):
            articles = fallback_search()
            if articles:
                models = extract_models_from_articles(articles)
                return {"status": "fallback", "models": models, "model_count": len(models), "articles": articles}
            result["error"] = "官网被反爬拦截"
            return result

        models = extract_from_md(md)
        if models:
            result["status"] = "success"
            result["models"] = models
            result["model_count"] = len(models)
        else:
            result["status"] = "partial"
            result["error"] = "页面抓取成功但未提取到产品"
            result["markdown_preview"] = md[:800]

        return result

    except Exception as e:
        result["error"] = str(e)[:200]
        articles = fallback_search()
        if articles:
            models = extract_models_from_articles(articles)
            return {"status": "fallback", "models": models, "model_count": len(models), "articles": articles}
        return result

def extract_models_from_articles(articles):
    """从搜索结果（标题+摘要）中提取型号"""
    seen = set()
    models = []

    text = ' '.join([
        (a.get('title', '') or '') + ' ' + (a.get('snippet', '') or '')
        for a in articles
    ])

    non_fridge_prefix = re.compile(
        r'^(MED|DV|WED|WGD|MVW|MVWB|WTW|WT|NTW|MHW|MHN|MGD|MLE|MLG|'
        r'MDB|MDF|MDT|MDP|WDF|WDT|WDP|KDT|KDF|KDPE|KDTE|SHX|SHP|SHE|'
        r'DW|DWV|DD|DP|GU|GC|GE|GP|GDT|GDF|GDP|GD|GF|GS|'
        r'MMV|ME|MS|MC|NN|NE|NS|NX|SN|SE|SG)', re.IGNORECASE)

    model_rx = [
        r'\b([A-Z]{2,6}[\-]?\d{2,6}[A-Z]{0,4}[A-Z0-9]{0,4})\b',
        r'\b([A-Z]{2,4}\d{2,4}[A-Z]{2,4})\b',
        r'Model\s*(?:NO\.|No|Number)?\s*[：:]*\s*([A-Z0-9\-]{6,15})',
    ]

    for rx in model_rx:
        for m in re.finditer(rx, text, re.IGNORECASE):
            model = m.group(1).strip()
            if 5 <= len(model) <= 20 and not model.isdigit():
                if re.match(r'^(HTTP|HTTPS|WWW|CSS|HTML|JS|API)', model, re.IGNORECASE):
                    continue
                if non_fridge_prefix.match(model):
                    continue
                if re.search(r'\d{2,4}x\d{2,4}', model, re.IGNORECASE):
                    continue
                if re.search(r'\d+inch', model, re.IGNORECASE):
                    continue
                if re.match(r'^[a-z]+-\d{6,}$', model, re.IGNORECASE):
                    continue
                if re.match(r'^SDSAC', model, re.IGNORECASE):
                    continue
                if model not in seen:
                    if not re.search(r'[A-Z]', model):
                        continue
                    if not re.search(r'\d', model):
                        continue
                    seen.add(model)
                    models.append({"name": model, "model": model})

    return models[:20]


def extract_from_md(md):
    seen = set()
    items = []

    noise = [
        'special offer', 'appliance package', 'skip to main', 'icon/error',
        'order status', 'contact us', 'register', 'service & repair',
        'replacement parts', 'every day, care', 'recommendations',
        'shop now', 'shop sales', 'shop closeout', 'view now',
        'remove all', 'check eligibility', 'learn more', 'alt text',
        'product ratings', 'number of ratings', 'track my order',
        'bundle', 'package', 'deal', 'sale', 'closeout', 'outlet',
        'made to work better', 'life is complicated', 'engineered for speed',
        'meet your culinary', 'elevate your culinary', 'enjoy easy',
        'power up your clean', 'brandlogo', 'brand logo',
        'facebook', 'twitter', 'youtube', 'instagram', 'pinterest',
        'privacy', 'terms', 'cookie', 'accessibility', 'sitemap',
        'warranty', 'delivery', 'help center', 'find a store',
        'compare', 'wishlist', 'chat with', 'sign in', 'log in',
        'create account', 'my account', 'cart', 'subscribe',
        'product information sheet', 'product help', 'homepage',
        'find the best', 'view all faq', 'click here', 'close icon',
        'mail icon', 'affirm.com', 'connects to your home',
        'washer', 'dryer', 'range', 'cooktop', 'hood', 'microwave',
        'dishwasher', 'vacuum', 'air conditioner', 'dehumidifier',
        'disposal', 'compactor', 'water heater',
    ]

    # 纯分类导航词（短名且不含品牌/容量/型号信息）
    nav_only = re.compile(
        r'^(french\s*door|side.by.side|top\s*freezer|'
        r'counter.depth\s*(models)?|smart\s*refrigerator|'
        r'refrigeration)$', re.IGNORECASE)

    for m in re.finditer(r'\[([^\]]+)\]\(([^)]+)\)', md):
        name = m.group(1).strip()
        url = m.group(2).strip()

        if re.search(r'\.(svg|png|jpg|gif|webp|ico)(\?|$)', name, re.IGNORECASE):
            continue
        if re.search(r'\$ORIGIN_IMG\$', name):
            continue
        if name.startswith('!'):
            continue

        name_lower = name.lower()

        if '\n' in name or '\\\\n' in name:
            continue

        if any(w in name_lower for w in noise):
            continue

        # 纯分类导航词（如"French door"但非"Bespoke AI 4-Door French Door"）
        if nav_only.match(name):
            continue

        if re.search(r'\*\*.*ratings', name, re.IGNORECASE):
            continue

        if re.search(r'(deals?|offer|sale|register|contact|support|service|parts|accessories|track|my-account|login|about|careers?|press|javascript)', url, re.IGNORECASE):
            continue

        if 8 <= len(name) <= 200:
            if name not in seen:
                seen.add(name)
                items.append({"name": name, "url": url})

    # 非冰箱品类型号前缀 (烘干机/洗衣机/洗碗机)
    non_fridge_prefix = re.compile(
        r'^(MED|DV|WED|WGD|MVW|MVWB|WTW|WT|NTW|MHW|MHN|MGD|MLE|MLG|'
        r'MDB|MDF|MDT|MDP|WDF|WDT|WDP|KDT|KDF|KDPE|KDTE|SHX|SHP|SHE|'
        r'DW|DWV|DD|DP|GU|GC|GE|GP|GDT|GDF|GDP|GD|GF|GS|'
        r'MMV|ME|MS|MC|NN|NE|NS|NX|SN|SE|SG)', re.IGNORECASE)

    model_rx = [
        r'\b([A-Z]{2,6}[\-]?\d{2,6}[A-Z]{0,4}[A-Z0-9]{0,4})\b',
        r'\b([A-Z]{2,4}\d{2,4}[A-Z]{2,4})\b',
        r'Model\s*(?:NO\.|No|Number)?\s*[：:]*\s*([A-Z0-9\-]{6,15})',
    ]
    for rx in model_rx:
        for m in re.finditer(rx, md, re.IGNORECASE):
            model = m.group(1).strip()
            if 5 <= len(model) <= 20 and not model.isdigit():
                if re.match(r'^(HTTP|HTTPS|WWW|CSS|HTML|JS|API)', model, re.IGNORECASE):
                    continue
                if non_fridge_prefix.match(model):
                    continue
                if re.search(r'\d{2,4}x\d{2,4}', model, re.IGNORECASE):
                    continue
                if re.search(r'\d+inch', model, re.IGNORECASE):
                    continue
                if re.match(r'^[a-z]+-\d{6,}$', model, re.IGNORECASE):
                    continue
                # 过滤三星内部图片ID (SDSAC-*)
                if re.match(r'^SDSAC', model, re.IGNORECASE):
                    continue
                if model not in seen:
                    # 真实型号至少含1个大写字母+1个数字，过滤URL路径片段
                    if not re.search(r'[A-Z]', model):
                        continue
                    if not re.search(r'\d', model):
                        continue
                    seen.add(model)
                    items.append({"name": model, "model": model})

    return items[:30]


def extract_amazon(md):
    products = []
    price_rx = r'(?:€|EUR|USD|\$)\s*[\d.,]+'
    lines = md.split('\n')
    cur = {}

    for line in lines:
        line = line.strip()
        if not line:
            if cur.get('name'):
                products.append(cur)
                cur = {}
            continue
        if re.match(r'^(Sponsored|Gesponsert|Results|Ergebnisse|Filter|Sort|Page|Seite)', line, re.IGNORECASE):
            continue
        if len(line) > 30 and not line.startswith(('http', '//', '(', '[')):
            if not cur.get('name'):
                cur['name'] = line[:150]
        pm = re.search(price_rx, line)
        if pm and not cur.get('price'):
            cur['price'] = pm.group(0)
        rm = re.search(r'(\d+[.,]?\d*)\s*(?:out of|von|de)\s*5\s*(?:stars|Sternen)', line)
        if rm and not cur.get('rating'):
            cur['rating'] = rm.group(1)

    if cur.get('name'):
        products.append(cur)
    return products[:10]


def filter_fresh(articles):
    this_year = datetime.now().year
    out = []
    for a in articles:
        text = a.get('title', '') + ' ' + a.get('snippet', '')
        years = re.findall(r'(20\d{2})', text)
        if years:
            if max(int(y) for y in years) >= this_year - 1:
                out.append(a)
        else:
            out.append(a)
    return out
